```go
// gitReadAttempts 번 모두 다르면 repository_changing. 기한은 재시도를 포함해 조회 하나 전체에 걸린다(같은 ctx).
// 경계 판정(parent_unavailable·relation·range_complete)은 그 한 번의 목록으로만 한다 — 명령마다 따로 방어하지 않는다.
func (reader *gitReader) readConsistently(read gitRead, shas []string) (any, error) {
	for attempt := 0; attempt < gitReadAttempts; attempt++ {
		path, err := reader.shallowPath()
		if err != nil {
			return nil, err
		}
		before, err := readShallowList(path)
		if err != nil {
			return nil, err
		}
		data, readErr := read(reader, shas, shallowSet(before))
		if errors.Is(readErr, errGitTimeout) {
			return nil, readErr
		}
		after, err := readShallowList(path)
		if err != nil {
			return nil, err
		}
		if after == before {
			return data, readErr
		}
	}
	return nil, errRepositoryChanging
}
```

```go
// relation — 명세 12.8 의 판정 순서. 얕은 이력에서는 A 객체가 남아 있어도 B 에서 A 까지의 이력이 경계에서 끊길 수 있어
// merge-base 의 "조상 아님"만으로는 되돌림·갈라짐이라 말할 수 없다.
//  1. A 가 B 의 조상 → ancestor
//  2. B 의 조상 순회가 얕은 경계에 닿지 않는다(이력이 온전하다) → not_ancestor
//  3. B 가 A 의 조상(확실한 되돌림) → not_ancestor
//  4. 그 밖 → unknown
func (reader *gitReader) relation(from, to string, boundaries map[string]bool) (string, error) {
	ancestor, err := reader.isAncestor(from, to)
	if err != nil || ancestor {
		return gitRelationAncestor, err
	}
	reached, err := reader.reachesShallowBoundary(to, boundaries)
	if err != nil {
		return "", err
	}
	if !reached {
		return gitRelationNotAncestor, nil
	}
	rolledBack, err := reader.isAncestor(to, from)
	if err != nil {
		return "", err
	}
	if rolledBack {
		return gitRelationNotAncestor, nil
	}
	return gitRelationUnknown, nil
}
```
